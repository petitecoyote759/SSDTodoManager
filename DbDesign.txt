Table books {
  book_id uuid [primary key, unique, default: 'gen_random_uuid()']
  user_id uuid [primary key]
  name varchar(255) [not null]
}

Table notes {
  note_id uuid [primary key, unique, default: 'gen_random_uuid()']
  book_id uuid [primary key]
  name varchar(255) [not null]
  colour varchar(15) [not null]
  white_text boolean [not null]
  posx float [not null]
  posy float [not null]
}

Table entries {
  entry_id uuid [primary key, unique, default: 'gen_random_uuid()']
  note_id uuid [primary key]
  text varchar(255) [not null]
  completion varchar(255) [not null]
  priority smallint [not null]
  due_date date
  description varchar(255) [not null]
}





Table users {
  user_id uuid [primary key, unique, default: 'gen_random_uuid()']
  username varchar(255) [not null]
  password_hash varchar(255) [not null]
  email varchar(255) [not null, unique]
  profile_picture varchar(255)
  created_at timestamp [not null, default: 'CURRENT_TIMESTAMP']
}




Ref: users.user_id <? books.user_id
Ref: books.book_id <? notes.book_id
Ref: notes.note_id <? entries.note_id
