CREATE TABLE "books" (
  "book_id" uuid UNIQUE DEFAULT gen_random_uuid(),
  "user_id" uuid,
  "name" varchar(255) NOT NULL,
  PRIMARY KEY ("book_id", "user_id")
);

CREATE TABLE "notes" (
  "note_id" uuid UNIQUE DEFAULT gen_random_uuid(),
  "book_id" uuid,
  "name" varchar(255) NOT NULL,
  "colour" varchar(15) NOT NULL,
  "white_text" boolean NOT NULL,
  "posx" float NOT NULL,
  "posy" float NOT NULL,
  PRIMARY KEY ("note_id", "book_id")
);

CREATE TABLE "entries" (
  "entry_id" uuid UNIQUE DEFAULT gen_random_uuid(),
  "note_id" uuid,
  "text" varchar(255) NOT NULL,
  "completion" varchar(255) NOT NULL,
  "due_date" date,
  "description" varchar(255) NOT NULL,
  PRIMARY KEY ("entry_id", "note_id")
);

CREATE TABLE "users" (
  "user_id" uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  "username" varchar(255) NOT NULL,
  "email" varchar(255) NOT NULL UNIQUE,
  "password_hash" varchar(255) NOT NULL,
  "profile_picture" varchar(255),
  "created_at" timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP
);



ALTER TABLE "books" ADD FOREIGN KEY ("user_id") REFERENCES "users" ("user_id") DEFERRABLE INITIALLY IMMEDIATE;

ALTER TABLE "notes" ADD FOREIGN KEY ("book_id") REFERENCES "books" ("book_id") DEFERRABLE INITIALLY IMMEDIATE;
ALTER TABLE "entries" ADD FOREIGN KEY ("note_id") REFERENCES "notes" ("note_id") DEFERRABLE INITIALLY IMMEDIATE;
